'absence-alert' : '' )} ${kpi( 'Desvíos', num(det.length), 'Día seleccionado' )} ${kpi( 'Líneas afectadas', num(lines.size), 'Líneas reales' )}

Novedades mecánicas ${num(inc.length)}

${table( [ [ 'Línea', 'linea' ], [ 'Interno', 'interno' ], [ 'Falla', 'falla', v => esc( normalizeMechanicalCategory(v) ) ], [ 'Intervención', 'intervencion' ], [ 'Resolución', 'resolucion', v => { if ( v && String(v).trim() ) { const pending = normalizeGuardText(v) .includes('PENDIENTE'); return ` ${esc(v)} `; } return ` Pendiente `; } ] ], inc )}

Reemplazos de conductores ${num(drv.length)}

Desvíos ${num(det.length)}

TURNO MAÑANA ${num( groupedObs.manana.length )} ${ groupedObs.manana.length === 1 ? 'novedad' : 'novedades' }
${renderGuardEvents( groupedObs.manana, 'Sin novedades registradas para el Turno Mañana.' )}
TURNO TARDE ${num( groupedObs.tarde.length )} ${ groupedObs.tarde.length === 1 ? 'novedad' : 'novedades' }
${renderGuardEvents( groupedObs.tarde, 'Sin novedades registradas para el Turno Tarde.' )}
`; }
