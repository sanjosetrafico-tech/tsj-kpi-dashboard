// Google Sheet: IPK POR LINEA SEP const IPK_SHEET_ID = '1jfqsPpANydPYXvvH7zkFeq-dnuPi0ogSD51sosRvQYc'; const KPI_TAB = 'PLAN'; const IPK_TAB = 'Ipk mensual'; // ===================================================== // ENDPOINT PRINCIPAL // ===================================================== function doGet(e) { try { const daily = obtenerKPIDiarios(); const targets = obtenerMetasAcumuladas(); const ipkMonthly = obtenerIPKMensualPorLinea(); const resultado = { ok: true, updatedAt: new Date() .toISOString(), data: { daily: daily, targets: targets, ipkMonthly: ipkMonthly } }; return responder( resultado, e ); } catch (error) { const resultado = { ok: false, error: error.message }; return responder( resultado, e ); } } // ===================================================== // RESPUESTA JSON / JSONP // ===================================================== function responder(data, e) { const callback = e && e.parameter && e.parameter.callback ? e.parameter.callback : null; // JSONP if (callback) { return ContentService .createTextOutput( callback + '(' + JSON.stringify(data) + ');' ) .setMimeType( ContentService .MimeType .JAVASCRIPT ); } // JSON normal return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService .MimeType .JSON ); } // ===================================================== // KPI DIARIOS // ===================================================== function obtenerKPIDiarios() { const ss = SpreadsheetApp .openById( KPI_SHEET_ID ); const sheet = ss.getSheetByName( KPI_TAB ); if (!sheet) { throw new Error( 'No existe la pestaña PLAN' ); } const lastRow = sheet.getLastRow(); const values = sheet .getRange( 3, 1, lastRow - 2, 25 ) .getDisplayValues(); const daily = {}; values.forEach( row => { const fechaTexto = String( row[0] || '' ).trim(); if ( !fechaTexto || fechaTexto .toUpperCase() === 'TOTAL' ) { return; } const fecha = convertirFecha( fechaTexto ); if (!fecha) { return; } // Si todavía no tiene // servicio ejecutado, // no consideramos cargado el día. const servicioEjecutado = row[2]; if ( servicioEjecutado === '' || servicioEjecutado === null ) { return; } // --------------------------------------------- // COLUMNAS DE PLAN // // D = Cumplimiento servicio // I = Cumplimiento km // L = Titularidad // O = Cumplimiento hs extras // P = Siniestros // R = IPK // S = Cumplimiento horario // T = Cabecera // U = Portón // V = Estética // X = Vueltas perdidas // --------------------------------------------- const service = numero( row[3] ); const kms = numero( row[8] ); const titularidad = numero( row[11] ); const extra = numero( row[14] ); const siniestros = numero( row[15] ); const ipk = numero( row[17] ); const horario = numero( row[18] ); const cabecera = numero( row[19] ); const porton = numero( row[20] ); const estetica = numero( row[21] ); const vueltas = numero( row[23] ); daily[fecha] = { service: crearPorcentaje( service, service >= 95 ), kms: crearPorcentaje( kms, kms >= 95 ), extra: { value: formatoPorcentaje( extra ), status: extra <= 100 ? 'good' : 'warn', progress: limitar( extra ) }, titularidad: crearPorcentaje( titularidad, titularidad >= 80 ), sinRes: { value: String( siniestros ), status: siniestros <= 10 ? 'good' : 'bad', progress: limitar( siniestros * 10 ) }, // Se completa luego // desde el resumen lateral. sinResNo: { value: '0', status: 'good', progress: 0 }, horario: crearPorcentaje( horario, horario >= 90 ), cabecera: crearPorcentaje( cabecera, cabecera >= 90 ), porton: crearPorcentaje( porton, porton >= 90 ), ipk: { value: formatoDecimal( ipk ), status: ipk >= 2 ? 'good' : 'bad', progress: limitar( (ipk / 2) * 100 ) }, estetica: crearPorcentaje( estetica, estetica >= 100 ), vueltas: { value: String( vueltas ), status: vueltas === 0 ? 'good' : 'warn', progress: limitar( vueltas * 10 ) } }; } ); actualizarUltimoDiaDesdeResumen( daily ); return daily; } // ===================================================== // ACTUALIZAR ÚLTIMO DÍA DESDE RESUMEN // ===================================================== function actualizarUltimoDiaDesdeResumen( daily ) { const ss = SpreadsheetApp .openById( KPI_SHEET_ID ); const sheet = ss.getSheetByName( KPI_TAB ); // AB:AF // // Indicador | fecha | Meta | Acum const values = sheet .getRange( 'AB4:AF20' ) .getDisplayValues(); if (!values.length) { return; } const fechaTexto = values[0][1]; const fecha = convertirFecha( fechaTexto ); if ( !fecha || !daily[fecha] ) { return; } for ( let i = 1; i < values.length; i++ ) { const indicador = normalizarTexto( values[i][0] ); const valor = values[i][1]; if (!indicador) { continue; } // Siniestros con responsabilidad if ( indicador.includes( 'siniestros con responsabilidad' ) ) { const n = numero( valor ); daily[fecha] .sinRes = { value: String(n), status: n <= 10 ? 'good' : 'bad', progress: limitar( n * 10 ) }; } // Siniestros sin responsabilidad if ( indicador.includes( 'siniestros sin respon' ) ) { const n = numero( valor ); daily[fecha] .sinResNo = { value: String(n), status: n <= 6 ? 'good' : 'bad', progress: limitar( n * 15 ) }; } } } // ===================================================== // METAS Y ACUMULADOS // ===================================================== function obtenerMetasAcumuladas() { const ss = SpreadsheetApp .openById( KPI_SHEET_ID ); const sheet = ss.getSheetByName( KPI_TAB ); const values = sheet .getRange( 'AB4:AF20' ) .getDisplayValues(); const resultado = []; if (!values.length) { return resultado; } for ( let i = 1; i < values.length; i++ ) { const nombre = values[i][0]; if (!nombre) { continue; } resultado.push({ indicador: nombre, valorDia: values[i][1], meta: values[i][2], acumulado: values[i][3] }); } // Cabecera y Portón // se obtienen de TOTAL. const lastRow = sheet.getLastRow(); const data = sheet .getRange( 1, 1, lastRow, 21 ) .getDisplayValues(); const total = data.find( row => String( row[0] ) .trim() .toUpperCase() === 'TOTAL' ); if (total) { resultado.push({ indicador: 'Cumplimiento Cabecera', valorDia: '', meta: '≥90%', acumulado: total[19] }); resultado.push({ indicador: 'Cumplimiento Portón', valorDia: '', meta: '≥90%', acumulado: total[20] }); } return resultado; } // ===================================================== // IPK MENSUAL POR LÍNEA // ===================================================== function obtenerIPKMensualPorLinea() { const ss = SpreadsheetApp .openById( IPK_SHEET_ID ); const sheet = ss.getSheetByName( IPK_TAB ); if (!sheet) { throw new Error( 'No existe la pestaña Ipk mensual' ); } const values = sheet .getDataRange() .getDisplayValues(); const lineas = [ 'A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'L', 'M', 'P' ]; const acumulado = {}; lineas.forEach( linea => { acumulado[linea] = { km: 0, pasajeros: 0 }; } ); // Busca bloques: // // LINEA | KM | PASAJEROS | IPK for ( let r = 0; r < values.length; r++ ) { for ( let c = 0; c < values[r].length; c++ ) { const header = String( values[r][c] || '' ).trim(); if ( !/^IPK\s+\d{1,2}-\d{1,2}$/i .test( header ) ) { continue; } for ( let rr = r + 1; rr <= r + 11 && rr < values.length; rr++ ) { const linea = String( values[rr][c - 3] || '' ).trim(); if ( !acumulado[linea] ) { continue; } const km = numero( values[rr][c - 2] ); const pasajeros = numero( values[rr][c - 1] ); acumulado[linea] .km += km; acumulado[linea] .pasajeros += pasajeros; } } } return lineas.map( linea => { const item = acumulado[linea]; const ipk = item.km > 0 ? item.pasajeros / item.km : 0; return { linea: linea, km: redondear( item.km, 2 ), pasajeros: Math.round( item.pasajeros ), ipk: redondear( ipk, 2 ) }; } ); } // ===================================================== // CREAR KPI PORCENTAJE // ===================================================== function crearPorcentaje( valor, cumple ) { return { value: formatoPorcentaje( valor ), status: cumple ? 'good' : 'bad', progress: limitar( valor ) }; } // ===================================================== // CONVERTIR TEXTO A NÚMERO // ===================================================== function numero(valor) { if ( valor === null || valor === undefined || valor === '' ) { return 0; } let texto = String( valor ) .trim() .replace( '%', '' ) .replace( /\s/g, '' ); // Ej: // 1,658 -> 1658 // // 60,5 -> 60.5 if ( /^\d{1,3}(,\d{3})+$/ .test( texto ) ) { texto = texto.replace( /,/g, '' ); } else { texto = texto.replace( ',', '.' ); } texto = texto.replace( /[^0-9.\-]/g, '' ); const n = Number( texto ); return Number.isFinite( n ) ? n : 0; } // ===================================================== // FORMATO PORCENTAJE // ===================================================== function formatoPorcentaje( valor ) { const n = Number( valor ); if ( !Number.isFinite( n ) ) { return '—'; } return ( formatoNumero( n ) + '%' ); } // ===================================================== // FORMATO DECIMAL // ===================================================== function formatoDecimal( valor ) { const n = Number( valor ); if ( !Number.isFinite( n ) ) { return '—'; } return n .toFixed( 2 ) .replace( '.', ',' ); } // ===================================================== // FORMATO NÚMERO // ===================================================== function formatoNumero( valor ) { const n = Number( valor ); if ( Number.isInteger( n ) ) { return String( n ); } return n .toFixed( 1 ) .replace( '.', ',' ); } // ===================================================== // CONVERTIR FECHA // ===================================================== function convertirFecha( valor ) { if (!valor) { return null; } const texto = String( valor ) .trim() .toLowerCase(); const match = texto.match( /^(\d{1,2})-([a-záéíóú]{3,})/ ); if (!match) { return null; } const dia = Number( match[1] ); const meses = { ene: 1, feb: 2, mar: 3, abr: 4, may: 5, jun: 6, jul: 7, ago: 8, sep: 9, oct: 10, nov: 11, dic: 12 }; const mes = meses[ match[2] .substring( 0, 3 ) ]; if (!mes) { return null; } const yyyy = 2026; return ( yyyy + '-' + String( mes ) .padStart( 2, '0' ) + '-' + String( dia ) .padStart( 2, '0' ) ); } // ===================================================== // LIMITAR ENTRE 0 Y 100 // ===================================================== function limitar( valor ) { const n = Number( valor ); if ( !Number.isFinite( n ) ) { return 0; } return Math.max( 0, Math.min( 100, n ) ); } // ===================================================== // REDONDEAR // ===================================================== function redondear( valor, decimales ) { const factor = Math.pow( 10, decimales ); return ( Math.round( valor * factor ) / factor ); } // ===================================================== // NORMALIZAR TEXTO // ===================================================== function normalizarTexto( texto ) { return String( texto || '' ) .trim() .toLowerCase() .normalize( 'NFD' ) .replace( /[\u0300-\u036f]/g, '' ); }
